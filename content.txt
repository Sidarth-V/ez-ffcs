      <animateMotion dur="16s" repeatCount="indefinite" path="M0,0 a98,98 0 1,0 0.001,0"/>
    </circle>
    <circle r="3" fill="var(--green)" opacity="0.6">
      <animateMotion dur="7s" repeatCount="indefinite" path="M0,0 a45,45 0 1,1 0.001,0"/>
    </circle>
  </svg>`;
}

// ──────────────────────────────────────────────
// RENDER ENGINE
// ──────────────────────────────────────────────
let current = 0;
const deck = document.getElementById('deck');

function renderAll() {
  deck.innerHTML = '';
  slides.forEach((s, i) => {
    const el = document.createElement('div');
    el.className = 'slide bg-grid' + (i === 0 ? ' active' : '');
    el.innerHTML = s.render();
    deck.appendChild(el);
  });
}

function goTo(idx) {
  const els = deck.querySelectorAll('.slide');
  els[current].classList.remove('active');
  els[current].classList.add('exit');
  setTimeout(() => els[current].classList.remove('exit'), 500);
  current = idx;
  els[current].classList.add('active');
  updateNav();
}

function updateNav() {
  const total = slides.length;
  document.getElementById('slide-num').textContent = `${current + 1} / ${total}`;
  document.getElementById('slide-title-nav').textContent = slides[current].title;
  document.getElementById('progress').style.width = `${((current + 1) / total) * 100}%`;
}

document.getElementById('btn-next').addEventListener('click', () => {
  if (current < slides.length - 1) goTo(current + 1);
});
document.getElementById('btn-prev').addEventListener('click', () => {
  if (current > 0) goTo(current - 1);
});
document.addEventListener('keydown', e => {
  if (e.key === 'ArrowRight' || e.key === 'Space') { if (current < slides.length - 1) goTo(current + 1); }
  if (e.key === 'ArrowLeft') { if (current > 0) goTo(current - 1); }
});

renderAll();
updateNav();
</script>
